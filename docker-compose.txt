version: '3.8'

services:
  postgres:
    image: postgres:16-alpine
    container_name: framework_practice_db
    restart: unless-stopped
    environment:
      POSTGRES_USER: app_user
      POSTGRES_PASSWORD: secret_password
      POSTGRES_DB: practice_db
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data

  redis:
    image: redis:7-alpine
    container_name: framework_practice_redis
    restart: unless-stopped
    command: redis-server --requirepass secret_redis_password
    ports:
      - "6379:6379"
    volumes:
      - redis_data:/data

  # --- НОВОЕ: PHP БЭКЕНД ---
  laravel_app:
    build:
      context: ../
      dockerfile: 01-docker/Dockerfile.laravel
    container_name: framework_practice_php
    restart: unless-stopped
    volumes:
      - ../laravel:/var/www/html # Монтируем папку с кодом для live-reload
    depends_on:
      - postgres
      - redis

  # --- НОВОЕ: ВЕБ-СЕРВЕР ---
  nginx:
    image: nginx:alpine
    container_name: framework_practice_nginx
    restart: unless-stopped
    ports:
      - "8080:80" # Наш сайт будет доступен на localhost:8080
    volumes:
      - ../laravel:/var/www/html # Nginx тоже должен видеть файлы
      - ../03-nginx/default.conf:/etc/nginx/conf.d/default.conf
    depends_on:
      - laravel_app
